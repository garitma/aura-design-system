```tsx
import * as React from "react"
import { Slot } from "@radix-ui/react-slot"
import { cva, type VariantProps } from "class-variance-authority"

import { cn } from "@/utils/class-names"
import { ButtonGroupContext } from "@/components/ui/Button"
import { Separator } from "@/components/ui/Separator"

/**
 * `default` maps child buttons to `md` / `icon-md` (52px).
 * `sm` maps them to `sm` / `icon-sm` (39px). `icon` is the same 39px control.
 */
type ButtonGroupSize = "default" | "sm"

const buttonGroupVariants = cva(
  "flex w-fit items-stretch [&>*]:focus-visible:z-10 [&>*]:focus-visible:relative [&>[data-slot=select-trigger]:not([class*='w-'])]:w-fit [&>input]:flex-1 has-[select[aria-hidden=true]:last-child]:[&>[data-slot=select-trigger]:last-of-type]:rounded-r-md has-[>[data-slot=button-group]]:gap-2",
  {
    variants: {
      orientation: {
        horizontal:
          "[&>*:not(:first-child)]:rounded-l-none [&>*:not(:first-child)]:border-l-0 [&>*:not(:last-child)]:rounded-r-none",
        vertical:
          "flex-col [&>*:not(:first-child)]:rounded-t-none [&>*:not(:first-child)]:border-t-0 [&>*:not(:last-child)]:rounded-b-none",
      },
    },
    defaultVariants: {
      orientation: "horizontal",
    },
  }
)

function ButtonGroup({
  className,
  orientation = "horizontal",
  size = "default",
  ...props
}: React.ComponentProps<"div"> &
  VariantProps<typeof buttonGroupVariants> & {
    size?: ButtonGroupSize
  }) {
  return (
    <ButtonGroupContext.Provider value={size}>
      <div
        role="group"
        data-slot="button-group"
        data-orientation={orientation}
        data-size={size}
        className={cn(buttonGroupVariants({ orientation }), className)}
        {...props}
      />
    </ButtonGroupContext.Provider>
  )
}

function ButtonGroupText({
  className,
  asChild = false,
  ...props
}: React.ComponentProps<"div"> & {
  asChild?: boolean
}) {
  const size = React.useContext(ButtonGroupContext)
  const Comp = asChild ? Slot : "div"

  return (
    <Comp
      className={cn(
        "flex items-center gap-0.5 rounded-sm border border-gray-6 bg-gray-2 px-2 font-medium [&_svg]:pointer-events-none",
        size === "sm" ? "h-3" : "h-4",
        className
      )}
      {...props}
    />
  )
}

function ButtonGroupSeparator({
  className,
  orientation = "vertical",
  ...props
}: React.ComponentProps<typeof Separator>) {
  return (
    <Separator
      data-slot="button-group-separator"
      orientation={orientation}
      className={cn(
        "relative !m-0 self-stretch bg-gray-6 data-[orientation=vertical]:h-auto",
        className
      )}
      {...props}
    />
  )
}

export {
  ButtonGroup,
  ButtonGroupSeparator,
  ButtonGroupText,
  buttonGroupVariants,
}
export type { ButtonGroupSize }
```

## Usage

`size` is `"default"` or `"sm"`. Default passes `md` (52px) and `icon-md` to the buttons. Small passes `sm` (39px) and `icon-sm`. `ButtonGroupText` is `h-4` or `h-3` to match.

`size="icon"` and `size="icon-sm"` are the 39px icon button. `size="icon-md"` is 52px. Icon-only buttons need an `aria-label`.

On a narrow screen, put groups in `flex flex-wrap gap-2` so they wrap.

Joined fill buttons keep an accent-10 edge between them. Inner corners are square; the outer corners keep `--aura-button-radius`.

### Small

```tsx
export const Small = () => (
  <div className="flex flex-wrap gap-2">
    <ButtonGroup size="sm" aria-label="Archive and report">
      <Button variant="pill">Archive</Button>
      <Button variant="pill">Report</Button>
    </ButtonGroup>
    <ButtonGroup size="sm" aria-label="Snooze">
      <Button variant="pill">Snooze</Button>
      <Button variant="pill" aria-label="More options">
        <DotsHorizontalIcon className="icon" />
      </Button>
    </ButtonGroup>
  </div>
)
```

### Button icon

A label button paired with a three-dots icon button, at the default size and at small.

```tsx
export const ButtonIcon = () => (
  <div className="flex flex-wrap gap-2">
    <ButtonGroup aria-label="Snooze">
      <Button variant="pill">Snooze</Button>
      <Button variant="pill" aria-label="More options">
        <DotsHorizontalIcon className="icon" />
      </Button>
    </ButtonGroup>
    <ButtonGroup size="sm" aria-label="Snooze small">
      <Button variant="pill">Snooze</Button>
      <Button variant="pill" aria-label="More options">
        <DotsHorizontalIcon className="icon" />
      </Button>
    </ButtonGroup>
  </div>
)
```

### Vertical

```tsx
export const Vertical = () => (
  <div className="flex flex-wrap gap-2">
    <ButtonGroup orientation="vertical" aria-label="Vertical button group">
      <Button variant="fill">One</Button>
      <Button variant="fill">Two</Button>
      <Button variant="fill">Three</Button>
    </ButtonGroup>
    <ButtonGroup
      orientation="vertical"
      size="sm"
      aria-label="Small vertical button group"
    >
      <Button variant="fill">One</Button>
      <Button variant="fill">Two</Button>
    </ButtonGroup>
  </div>
)
```
